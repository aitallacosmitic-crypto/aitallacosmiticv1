    function computeOfferTotal(unitPrice, qty, offers) {
      const base = unitPrice * qty;
      if (!offers) return { total: base, saved: 0, label: "" };

      let best = { total: base, saved: 0, label: "" };

      Object.keys(offers).forEach((k) => {
        const q = Number(k);
        if (qty >= q) {
          const off = offers[q];
          let total = base;

          if (off.type === "percent") total = base * (1 - off.value / 100);

          if (total < best.total) {
            best = { total, saved: base - total, label: off.label || "" };
          }
        }
      });

      return best;
    }
